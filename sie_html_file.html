Click or press Space to jump!

import pygame
import random

pygame.init()
screen = pygame.display.set_mode((500, 200))
pygame.display.set_caption("Sreekruth Vardhan - Python Game")
clock = pygame.time.Clock()

player_y = 150
player_vy = 0
gravity = 0.5
is_jumping = False
running = True

while running:
    screen.fill((30, 30, 36))
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_SPACE and not is_jumping:
                player_vy = -9
                is_jumping = True

    player_vy += gravity
    player_y += player_vy
    if player_y >= 150:
        player_y = 150
        is_jumping = False

    pygame.draw.rect(screen, (76, 209, 55), (50, player_y, 25, 25))
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
